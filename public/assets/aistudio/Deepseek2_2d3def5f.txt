        url = f"https://fred.stlouisfed.org/graph/fredgraph.csv?id={series_id}"
        try:
            res = requests.get(url, timeout=10)
            if res.status_code == 200:
                df = pd.read_csv(io.StringIO(res.text))
                df.columns = ['Date', series_id]
                df['Date'] = pd.to_datetime(df['Date'], utc=True)
                df.set_index('Date', inplace=True)
                return df[series_id].replace('.', np.nan).astype(float)
        except Exception: pass

async function loadData(){
  const btn = document.getElementById('refreshBtn');
  const size = parseFloat(document.getElementById('accountSize').value) || 1000;
  btn.disabled = true;
  setStatus('loading','Waking up the Render backend... This can take up to 90 seconds.');
  const ctrl = new AbortController();
  const timer = setTimeout(()=>ctrl.abort(), TIMEOUT);
  try {
    const res = await fetch(BACKEND_URL + '/api/gold-predictions',{
      method:'POST',
      headers:{'Content-Type':'application/json'},
      body:JSON.stringify({account_size:size}),
      signal:ctrl.signal
    });
    clearTimeout(timer);
    let data;
    try { data = await res.json(); } catch { throw new Error('Invalid JSON (status '+res.status+')'); }
    if (!res.ok) { setStatus('error','⚠️ '+(data.error||('HTTP '+res.status))); return; }
    render(data);
    if (data.has_synthetic) setStatus('warn','⚠️ Some data sources fell back to synthetic. Reduce size.');
    else setStatus('ok','✅ All sources healthy. Signal generated.');
  } catch(e){
    clearTimeout(timer);
    if (e.name === 'AbortError') setStatus('error','⏱️ Timed out after 2 minutes. Try again.');
    else setStatus('error','❌ '+e.message+' — check BACKEND_URL.');
  } finally { btn.disabled = false; }
}

  push:

        default: 'GitHub UI'

jobs:
  patch:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Apply All Patches
        run: |
          for f in patch*.py; do 
            [ -e "$f" ] || continue
            echo "Running $f"
            python "$f"
          done
      - name: Commit if changed
        run: |
          git config user.name "ceo-bot"
          git config user.email "ceo-bot@masterpiece"
          git add -A
          git diff --cached --quiet && exit 0
          git commit -m "auto: apply patches [skip ci]"
          git push
  deploy:
    needs: patch
    runs-on: ubuntu-latest
    steps:
      - name: Trigger Render
        run: curl -fsS -X POST "${{ secrets.RENDER_DEPLOY_HOOK }}"
      - name: Verify Backend
        run: sleep 45 && curl -fsS "${{ secrets.BACKEND_URL }}/version" || exit 1
  apk:
    needs: deploy
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - uses: actions/setup-java@v4
        with: { distribution: 'temurin', java-version: '21' }
      - name: Build Android APK with Kotlin Exclusion
        run: |
          npm init -y
          npm install @capacitor/core @capacitor/cli @capacitor/android
          echo '{"appId":"com.masterpiece.xauusd","appName":"XAUUSD Masterpiece","webDir":"www"}' > capacitor.config.json
          mkdir -p www && cp index.html www/
          npx cap add android && npx cap sync android
          cat >> android/build.gradle << 'EOF'
          allprojects {
              configurations.all {
                  resolutionStrategy {
                      force 'org.jetbrains.kotlin:kotlin-stdlib:1.8.22'

          EOF
          echo "org.gradle.jvmargs=-Xmx2048m -Dfile.encoding=UTF-8" >> android/gradle.properties
      - name: Build APK
        run: cd android && chmod +x gradlew && ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: XAUUSD-Masterpiece-APK
          path: android/app/build/outputs/apk/debug/app-debug.apk
