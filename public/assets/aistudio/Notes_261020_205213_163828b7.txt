```python
from datetime import datetime, timezone
from flask import Flask, jsonify, request
from flask_cors import CORS
import numpy as np
import pandas as pd
import requests
import io
import yfinance as yf
from scipy.stats import t as scipy_t
import xgboost as xgb
from sklearn.ensemble import RandomForestClassifier, VotingClassifier
from sklearn.calibration import CalibratedClassifierCV
from sklearn.model_selection import TimeSeriesSplit
import concurrent.futures
import time
import traceback
import warnings
warnings.filterwarnings('ignore')

app = Flask(__name__)
CORS(app)

MODEL_CACHE = {"model": None, "last_trained": 0}
CACHE_TIMEOUT = 900

class DataOrchestrator:
    def __init__(self):
        self.redundancy_report = {}
        self.headers = {'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36'}

    def _log(self, feature, source_name, tier, status):
        self.redundancy_report[feature] = {"source": source_name, "tier": tier, "status": status}

    def fetch_fred_csv(self, series_id):
        url = f"https://fred.stlouisfed.org/graph/fredgraph.csv?id={series_id}"
        try:
            res = requests.get(url, timeout=10)
            if res.status_code == 200:
                df = pd.read_csv(io.StringIO(res.text))
                df.columns = ['Date', series_id]
                df['Date'] = pd.to_datetime(df['Date'], utc=True)
                df.set_index('Date', inplace=True)
                return df[series_id].replace('.', np.nan).astype(float)
        except Exception: pass
        raise ValueError(f"FRED failed for {series_id}")

    def get_series(self, feature_name, sources, synth_val=0.0, freq='15min'):
        for src in sources:
            try:
                data = src['func']()
                if data is not None and not data.empty:
                    self._log(feature_name, src['name'], src['tier'], "HEALTHY")
                    return data
            except Exception: pass
        self._log(feature_name, "SYNTHETIC", 4, "CRITICAL")
        dates = pd.date_range(end=pd.Timestamp.now(tz='UTC'), periods=500, freq=freq)
        return pd.Series(synth_val, index=dates)

def get_gold_15m(orch):
    def binance():
        res = requests.get("https://api.binance.com/api/v3/klines?symbol=PAXGUSDT&interval=15m&limit=1000", timeout=5).json()
        return pd.Series([float(x[4]) for x in res], index=pd.to_datetime([int(x[0]) for x in res], unit='ms', utc=True)).dropna()
    return orch.get_series("Gold_15m", [{'name': 'BINANCE', 'tier': 1, 'func': binance}], synth_val=2000.0)

def get_macro(orch):
    def fred(series_id): return orch.fetch_fred_csv(series_id)
    try:
        dxy = fred("DTWEXBGS")
        real_yield = fred("DFII10")
        breakeven = fred("T10YIE")
        liquidity = fred("WALCL")
        vix = fred("VIXCLS")
        df = pd.DataFrame({"DXY": dxy, "Real_Yield": real_yield, "Breakeven": breakeven, "Liquidity": liquidity, "VIX": vix}).dropna()
        if df.empty: raise ValueError("Empty Macro Data")
        return df
    except Exception:
        dates = pd.date_range(end=pd.Timestamp.now(tz='UTC'), periods=500, freq='1d')
        return pd.DataFrame({"DXY": 104.0, "Real_Yield": 1.5, "Breakeven": 2.2, "Liquidity": 7000000.0, "VIX": 18.0}, index=dates)

def get_gdx(orch):
    def yahoo():
        df = yf.download("GDX", period="5y", interval="1d", progress=False)['Close']
        if isinstance(df, pd.DataFrame): df = df.iloc[:, 0]
        df.index = pd.to_datetime(df.index, utc=True)
        return df.dropna()
    return orch.get_series("GDX", [{'name': 'YAHOO', 'tier': 2, 'func': yahoo}], synth_val=30.0, freq='1d')

@app.route("/api/gold-predictions", methods=["POST"])
def get_predictions():
    global MODEL_CACHE
    try:
        req_data = request.get_json() or {}
        account_size = float(req_data.get("account_size", 1000))
        orch = DataOrchestrator()
        
        with concurrent.futures.ThreadPoolExecutor() as executor:
            f_gold_15m = executor.submit(get_gold_15m, orch)
            f_macro = executor.submit(get_macro, orch)
            f_gdx = executor.submit(get_gdx, orch)
            gold_15m = f_gold_15m.result()
            macro_df = f_macro.result()
            gdx = f_gdx.result()
        
        macro_df.index = pd.to_datetime(macro_df.index, utc=True).normalize() 
        gdx.index = pd.to_datetime(gdx.index, utc=True).normalize()
        
        df = pd.DataFrame({"Gold": gold_15m})
        df.index = pd.to_datetime(df.index, utc=True)
        df = df.sort_index()
        macro_df = macro_df.sort_index()
        gdx = gdx.sort_index()
        
        df = pd.merge_asof(df, macro_df, left_index=True, right_index=True, direction='backward')
        df = pd.merge_asof(df, gdx.rename("GDX"), left_index=True, right_index=True, direction='backward')
        
        df.dropna(inplace=True)
        df.replace([np.inf, -np.inf], np.nan, inplace=True)
        df.dropna(inplace=True)
        
        # 🚨 SAFETY CHECK: PREVENT TRAINING ON EMPTY DATA 🚨
        if len(df) < 200:
            return jsonify({"error": "Data Orchestrator failed to fetch sufficient data. Try again later."}), 500

        df['Real_Yield_Change'] = df['Real_Yield'].diff()
        df['Breakeven_Change'] = df['Breakeven'].diff()
        df['DXY_Change'] = df['DXY'].pct_change()
        df['Liquidity_Trend'] = df['Liquidity'].pct_change(periods=30)
        df['Miner_Divergence'] = df['GDX'].pct_change() - df['Gold'].pct_change()
        
        df['Returns_15m'] = df['Gold'].pct_change()
        delta = df['Gold'].diff()
        gain = (delta.where(delta > 0, 0)).rolling(14).mean()
        loss = (-delta.where(delta < 0, 0)).rolling(14).mean()
        df['RSI_15m'] = 100 - (100 / (1 + (gain / loss)))
        df['ATR_15m'] = (df['Gold'] - df['Gold'].shift(1)).abs().rolling(14).mean()
        
        current_candle_range = df['Gold'].iloc[-1] - df['Gold'].iloc[-2]
        atr_15m = float(df['ATR_15m'].iloc[-1])
        is_spread_too_wide = abs(current_candle_range) > (atr_15m * 0.5) 
        
        df['Target'] = (df['Gold'].shift(-5) > df['Gold']).astype(int)
        df.dropna(inplace=True)
        
        # 🚨 SAFETY CHECK 2: PREVENT TIME SERIES SPLIT ERROR 🚨
        if len(df) < 100:
            return jsonify({"error": "Not enough valid rows after feature engineering."}), 500

        features = ['Real_Yield_Change', 'Breakeven_Change', 'DXY_Change', 'Liquidity_Trend', 'Miner_Divergence', 'RSI_15m', 'Returns_15m']
        current_time = time.time()
        
        if MODEL_CACHE["model"] is None or (current_time - MODEL_CACHE["last_trained"]) > CACHE_TIMEOUT:
            X = df[features].iloc[-3000:] 
            y = df['Target'].iloc[-3000:]
            
            clf1 = xgb.XGBClassifier(n_estimators=50, max_depth=4, random_state=42)
            clf2 = RandomForestClassifier(n_estimators=50, max_depth=5, random_state=42)
            eclf = VotingClassifier(estimators=[('xgb', clf1), ('rf', clf2)], voting='soft')
            
            tscv = TimeSeriesSplit(n_splits=5, gap=5) 
            model = CalibratedClassifierCV(eclf, method='isotonic', cv=tscv)
            model.fit(X, y)
            
            MODEL_CACHE["model"] = model
            MODEL_CACHE["last_trained"] = current_time
        
        prob_up = float(MODEL_CACHE["model"].predict_proba(df[features].iloc[-1:])[0][1])
        current_price = float(df['Gold'].iloc[-1])
        
        macro_confluence = 0
        if df['Real_Yield_Change'].iloc[-1] < 0: macro_confluence += 25
        if df['Breakeven_Change'].iloc[-1] > 0: macro_confluence += 25
        if df['Miner_Divergence'].iloc[-1] > 0: macro_confluence += 25
        if df['DXY_Change'].iloc[-1] < 0: macro_confluence += 25

        current_hour_utc = datetime.now(timezone.utc).hour
        is_tradeable_session = 8 <= current_hour_utc <= 16
        action = "WAIT"
        risk_pct = 0.0
        sl = 0
        tp = 0
        max_risk = 0.5 if account_size < 1000 else 5.0

        if is_spread_too_wide:
            action = "WAIT (SPREAD TOO WIDE)"
        elif not is_tradeable_session:
            action = "WAIT (OUTSIDE SESSION)"
        elif macro_confluence >= 75 and prob_up > 0.60 and df['RSI_15m'].iloc[-1] < 60:
            action = f"INTRADAY BUY (Macro {macro_confluence}/100)"
            risk_pct = min(5.0 if macro_confluence >= 95 else 2.0, max_risk)
            sl = round(current_price - (1.5 * atr_15m), 2)
            tp = round(current_price + (2.5 * atr_15m), 2)
        elif macro_confluence <= 25 and prob_up < 0.40 and df['RSI_15m'].iloc[-1] > 40:
            action = f"INTRADAY SELL (Macro {macro_confluence}/100)"
            risk_pct = min(5.0 if macro_confluence <= 5 else 2.0, max_risk)
            sl = round(current_price + (1.5 * atr_15m), 2)
            tp = round(current_price - (2.5 * atr_15m), 2)

        last_time = int(df.index[-1].timestamp())
        time_step = 900 
        variance = df['Returns_15m'].var()
        theta = 0.15
        mu = df['Gold'].rolling(50).mean().iloc[-1]
        n_sims, n_steps = 500, 20
        simulations = np.zeros((n_sims, n_steps))
        random_shocks = scipy_t.rvs(df=3, size=(n_sims, n_steps))
        
        for i in range(n_sims):
            simulations[i, 0] = current_price
            for j in range(1, n_steps):
                reversion = theta * (mu - simulations[i, j-1])
                simulations[i, j] = simulations[i, j-1] + (simulations[i, j-1] * random_shocks[i, j] * np.sqrt(variance)) + reversion

        u95 = np.percentile(simulations, 95, axis=0)
        u75 = np.percentile(simulations, 75, axis=0)
        med = np.percentile(simulations, 50, axis=0)
        l25 = np.percentile(simulations, 25, axis=0)
        l05 = np.percentile(simulations, 5, axis=0)

        upper, upper_75, mid, lower_25, lower = [], [], [], [], []
        for i in range(n_steps):
            ft = last_time + ((i+1) * time_step)
            upper.append({"time": ft, "value": round(u95[i], 2)})
            upper_75.append({"time": ft, "value": round(u75[i], 2)})
            mid.append({"time": ft, "value": round(med[i], 2)})
            lower_25.append({"time": ft, "value": round(l25[i], 2)})
            lower.append({"time": ft, "value": round(l05[i], 2)})

        candles = []
        for i in range(len(df)):
            p = float(df['Gold'].iloc[i])
            op = float(df['Gold'].iloc[i-1]) if i > 0 else p
            candles.append({"time": int(df.index[i].timestamp()), "open": round(op, 2), "high": round(max(op, p) + 0.2, 2), "low": round(min(op, p) - 0.2, 2), "close": round(p, 2), "volume": 1000})

        return jsonify({
            "status": "SUCCESS",
            "current_price": current_price,
            "ml_prob_up": round(prob_up * 100, 1),
            "trade_setup": {
                "action": action, "confluence_score": macro_confluence,
                "risk_pct": risk_pct, "hard_stop_loss": sl, "take_profit": tp,
                "pyramiding_logic": "Disabled for 15m scalps"
            },
            "forecast": {"upper": upper, "upper_75": upper_75, "mid": mid, "lower_25": lower_25, "lower": lower},
            "candles": candles,
            "has_synthetic": any(info['tier'] == 4 for info in orch.redundancy_report.values()),
            "redundancy_report": orch.redundancy_report
        })
    except Exception as e:
        traceback_str = traceback.format_exc()
        return jsonify({"error": f"Internal Server Error: {str(e)}", "traceback": traceback_str}), 500

@app.route("/version", methods=["GET"])
```
